

RLS  prisma tables access

--RLS pour l acces avec prisma au tables

-- Policy to allow authenticated users to insert their own records
CREATE POLICY "Users can insert their own profileS" 
ON "User" 
FOR INSERT 
WITH CHECK (auth.uid()::text = id);

-- Policy to allow users to view only their own or public data
CREATE POLICY "Users can view limited user datas" 
ON "User" 
FOR SELECT 
USING (auth.uid()::text = id);


-- Enable RLS for Course table
ALTER TABLE "Course" ENABLE ROW LEVEL SECURITY;


CREATE POLICY "Teachers and admins can insert courses" 
ON "Course" 
FOR INSERT 
WITH CHECK (
  (SELECT role FROM "User" WHERE id = auth.uid()::text) IN ('TEACHER', 'ADMIN')
);

### Politiques RLS storage "avatar"


-- Autoriser les utilisateurs authentifiés à télécharger des fichiers dans le bucket "avatars"
CREATE POLICY "Allow authenticated users to upload"
ON storage.objects
FOR INSERT
WITH CHECK (auth.uid() IS NOT NULL);


-- Autoriser les utilisateurs authentifiés à lire leurs propres fichiers
CREATE POLICY "Allow users to read their own files"
ON storage.objects
FOR SELECT
USING (auth.uid()::text = owner_id);

-- Autoriser les utilisateurs authentifiés à mettre à jour leurs propres fichiers
CREATE POLICY "Allow users to update their own files"
ON storage.objects
FOR UPDATE
USING (auth.uid()::text = owner_id);

-- Autoriser les utilisateurs authentifiés à supprimer leurs propres fichiers
CREATE POLICY "Allow users to delete their own files"
ON storage.objects
FOR DELETE
USING (auth.uid()::text = owner_id);
